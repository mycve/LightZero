# CMakeLists.txt for C++ MCTS with libtorch
#
# 编译步骤：
# 1. 下载 libtorch: https://pytorch.org/get-started/locally/
# 2. 设置 Torch_DIR 环境变量
# 3. mkdir build && cd build
# 4. cmake -DCMAKE_PREFIX_PATH=/path/to/libtorch ..
# 5. make -j8

cmake_minimum_required(VERSION 3.14)
project(mcts_with_model LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 查找 PyTorch/libtorch
find_package(Torch REQUIRED)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${TORCH_CXX_FLAGS}")

# 查找 Python (用于 pybind11)
find_package(Python3 COMPONENTS Interpreter Development REQUIRED)
find_package(pybind11 REQUIRED)

# 源文件
set(SOURCES
    lib/cnode.cpp
    lib/cnode_with_model.cpp
    ../common_lib/cminimax.cpp
)

# 创建 Python 模块
pybind11_add_module(gmz_tree_cpp ${SOURCES} bindings.cpp)

# 链接 libtorch
target_link_libraries(gmz_tree_cpp PRIVATE "${TORCH_LIBRARIES}")

# 包含目录
target_include_directories(gmz_tree_cpp PRIVATE 
    ${CMAKE_CURRENT_SOURCE_DIR}/lib
    ${CMAKE_CURRENT_SOURCE_DIR}/../common_lib
)

# CUDA 支持
if(TORCH_CUDA_LIBRARIES)
    target_link_libraries(gmz_tree_cpp PRIVATE ${TORCH_CUDA_LIBRARIES})
endif()

# 设置 RPATH
set_property(TARGET gmz_tree_cpp PROPERTY 
    INSTALL_RPATH "${TORCH_INSTALL_PREFIX}/lib")
